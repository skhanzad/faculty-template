%% =====================================================================
%%  FACULTY APPLICATION PACKAGE
%%  Cover Letter . Curriculum Vitae . Research Statement . Teaching Dossier
%%  . EDIA Statement . References
%% =====================================================================
%%
%%  QUICK START
%%    1. Open config.tex and enter your details and the position.
%%    2. Edit the files in the documents/ folder.
%%    3. Put your publications in publications.bib.
%%    4. Recompile. Overleaf runs Biber and repeats the compile for you.
%%
%%  Delete the `tips' option below to hide the writing tips (the red
%%  boxes) before you submit.
%% =====================================================================

\documentclass[
  letterpaper,          % letterpaper (North America) | a4paper
  11pt,                 % 10pt | 11pt | 12pt
  font   = charter,     % charter | libertinus | palatino | sourcesans | lmodern
  accent = navy,        % navy | teal | burgundy | forest | slate | plum | ochre
                        % | black, or any xcolor colour, e.g. accent=blue!60!black
  header = split,       % split | centered  (letterhead layout)
  language = english,   % any babel language, e.g. canadian, british, french
  tips,                 % show writing tips in the PDF. DELETE before you submit
]{facultyapp}

\input{config}                      % <- your details and the position
\addbibresource{publications.bib}   % <- your publications (used in the CV)
